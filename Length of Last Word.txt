class Solution {
    public int lengthOfLastWord(String s) {
      if (s == null) {
            return 0;
        }

        String trimmedS = s.trim(); 
        int len = 0;

        for (int i = trimmedS.length() - 1; i >= 0; i--) {
            if (trimmedS.charAt(i) == ' ') {
                if (len > 0) { 
                    break;
                }
            } else {
                len++; // Increment counter for non-space characters
            }
        }
        return len;  
    }
}
